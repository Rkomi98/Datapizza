from datapizza_ai import Agent, Memory

# Agent with memory
agent = Agent(
    name="SmartBot",
    model="gpt-4",
    memory=Memory(type="conversation")
)

# It remembers context!
agent.chat("My name is Alex")
response = agent.chat("What's my name?")
# Output: "Your name is Alex!"

Power Up: Add Tools 🛠️

            

                
from datapizza_ai import Agent, Tool

# Define a custom tool
@Tool
def search_web(query: str):
    """Search the web for information"""
    return f"Results for: {query}"

# Agent with tools
agent = Agent(
    name="ResearchBot",
    tools=[search_web]
)

response = agent.chat("Find info about MAKEathon")
# Agent automatically uses the search tool!
